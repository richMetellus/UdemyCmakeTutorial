# Typical Manual way to build cmake project
#   0.) Create Source and CMakeFile
#   1.) mkdir build
#   2.) cd build
#   3.) cmake ..   -  Configure the Project -> Generate the Build Files (default Makefiles)
#   4.) cmake --build . 
#   5.) ./app/MyExecutable

cmake_minimum_required(VERSION 3.22)

project(
    CppProjectTemplate 
    VERSION 1.0.0 
    LANGUAGES C CXX)


# Overriding CMake Built-in Variable 
#------------------------------------
set(CMAKE_CXX_STANDARD          17) #override the compiler default standard
set(CMAKE_CXX_STANDARD_REQUIRED ON) #required a compiler that has fully implemented the C++ standard 17
set(CMAKE_CXX_EXTENSIONS        OFF) #turn off GCC C++ extensions
set(CMAKE_EXPORT_COMPILE_COMMANDS ON) # will help clang-tidy know which files belong to your code

# Custom Variable names
#--------------------------
# variable name      value
#         v            v
set(MY_LIB_NAME "MyLibrary") # quotation marks are optional, cmake can do automatic type
set(MY_EXECUTABLE_NAME MyExecutable)

# User defined CMake Options
# -------------------------------
# with options think boolean
#       option_variable name        optional description              default value
option(CRS_IS_COMPILE_EXECUTABLE_SET "Whether to compile the executable" ON)
option(CRS_ENABLE_TESTING "Enable a Unit Testing Build" ON)
option(CRS_ENABLE_WARNINGS "Enable compiler warnings" ON) # This should always be enabled
option(CRS_ENABLE_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF) # this option is up to you..too strict

option(CRS_ENABLE_SANITIZE_ADDR "Enable address sanitizer" OFF)
option(CRS_ENABLE_SANITIZE_UNDEF "Enable sanitize undefined" ON)
option(CRS_ENABLE_SANITIZE_LEAK "Enable leak sanitize (Gcc/Clang only)." OFF)
option(CRS_ENABLE_SANITIZE_THREAD "Enable thread sanitize (Gcc/Clang only)." OFF)
option(CRS_ENABLE_LTO "Enable Linked Time Optimization" ON)
option(CRS_ENABLE_CLANG_TIDY "Enable to add clang tidy." ON)

option(CRS_ENABLE_CMAKE_FORMAT "Enable to add cmake-format." OFF)
option(CRS_ENABLE_CLANG_FORMAT "Enable to add clang-format." OFF)

option(CRS_USE_CPM "Whether to use CPM as the package manager" ON)
option(CRS_USE_CONAN "Whether to use Conan as the package manager" OFF)
option(CRS_USE_VCPKG "Whether to use VCPKG" OFF)

# CMAKE MODULES
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake/")
include(CrsConfigSafeGuards)
#. Load and run our custom cmake module 
include(CrsAddGitSubmodule)
#. Load and run cmake built-in module FetchContent
include(FetchContent)
#. Load and run custom target to build documentation 
include(CrsDocs)
include(CrsTools)

if(CRS_ENABLE_WARNINGS)
    include(CrsWarnings)
endif()

crs_add_cmake_format_target()
crs_add_clang_format_target()

if(${CRS_ENABLE_SANITIZE_ADDR} OR ${CRS_ENABLE_SANITIZE_UNDEF})
    include(CrsSanitizers)
    crs_add_sanitizer_flags()
endif()

if(CRS_ENABLE_LTO)
    include(CrsLTO)
endif()

#. call cmake built-in module functions
#. Clone the github Cmake project nlohman_json non-recursively
#. Note: Project has to be a cmake project

if(CRS_USE_CPM)
    message(STATUS "using CPM")
    include(CPM)
    cpmaddpackage("gh:nlohmann/json#v3.11.3")
    cpmaddpackage("gh:fmtlib/fmt#9.1.0")
    cpmaddpackage("gh:gabime/spdlog#v1.13.0")
    cpmaddpackage("gh:jarro2783/cxxopts#v3.1.1")
    if(CRS_ENABLE_TESTING)
        cpmaddpackage("gh:catchorg/Catch2#v3.5.3")
        list(APPEND CMAKE_MODULE_PATH ${Catch2_SOURCE_DIR}/extras)
    endif()

elseif(CRS_USE_CONAN OR CRS_USE_VCPKG)
    if (CRS_USE_CONAN)
        message(STATUS "Using Conan")
        include(${CMAKE_BINARY_DIR}/conan_toolchain.cmake) # file generated by conan after a manual command
    elseif(CRS_USE_VCPKG)
        message(STATUS "Using VCPKG")
        include(${CMAKE_SOURCE_DIR}/external/vcpkg/scripts/buildsystems/vcpkg.cmake) 
    endif()
    find_package(nlohmann_json REQUIRED)
    find_package(fmt REQUIRED)
    find_package(spdlog REQUIRED)
    find_package(cxxopts REQUIRED)
    if(CRS_ENABLE_TESTING)
        find_package(Catch2 REQUIRED)
        list(APPEND CMAKE_MODULE_PATH ${Catch2_SOURCE_DIR}/extras)
    endif()
else()
    message(STATUS "using FetchContent")
    FetchContent_Declare(
        nlohman_json
        GIT_REPOSITORY https://github.com/nlohmann/json.git
        GIT_TAG v3.11.3
        GIT_SHALLOW TRUE #pass the ``--depth 1`` to git to only clone the recent commit tag, save bandwidth
    )
    #. Load the project into our build project
    FetchContent_MakeAvailable(nlohman_json)

    FetchContent_Declare(
        fmt
        GIT_REPOSITORY https://github.com/fmtlib/fmt
        GIT_TAG 9.1.0
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(fmt)

    FetchContent_Declare(
        spdlog
        GIT_REPOSITORY https://github.com/gabime/spdlog
        GIT_TAG v1.13.0
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(spdlog)

    FetchContent_Declare(
        cxxopts
        GIT_REPOSITORY https://github.com/jarro2783/cxxopts
        GIT_TAG v3.1.1
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(cxxopts)

    if(CRS_ENABLE_TESTING)
        FetchContent_Declare(
            Catch2
            GIT_REPOSITORY https://github.com/catchorg/Catch2
            GIT_TAG v3.5.3
            GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(Catch2)
        list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
    endif()

endif()

#. Order of listing will cause a build issue if add_subdirectory(inc) comes before src. 
# it customary to list the subdirectory 
# as per the order the target are build. Dependencies first
#. in inc folder we are using the target, so the target MyLibrary must be defined 
# first, that's why add_subdirectory(inc) is 2nd.
#. If the target is not defined first, how can we use the target? so the Order here matter
add_subdirectory(src)
add_subdirectory(inc)
add_subdirectory(config)

if (CRS_IS_COMPILE_EXECUTABLE_SET)
    add_subdirectory(app)
else()
    message("Building the project without creating the executable")
endif()

if(CRS_ENABLE_TESTING)
    include(CTest)
    enable_testing()
    add_subdirectory(tests)
endif()
